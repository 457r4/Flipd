WINDOW *container;

char track[512];

TUI::TUI(const Session &session) : session_(session) {}

void TUI::start() {

  initscr();
  cbreak();
  noecho();
  curs_set(0);

  start_color();
  use_default_colors();

  init_pair(2, COLOR_BLUE, -1);
  init_pair(10, COLOR_BLACK, -1);
  init_pair(11, COLOR_WHITE, COLOR_BLACK);

  thread music(&TUI::listener, this);
  music.detach();

  drawLayout();
  wattron(container, COLOR_PAIR(2));
  thread th(Time::run, this, &session_);
  th.detach();

  // cout << "Here is past launching thread" << endl;

  // popup = newwin(40, 40, 2, 2);
  // box(popup, 0, 0);
  // mvwprintw(popup, 20, 10, "Generic message");
  // wrefresh(popup);

  int ch;
  bool running = true;

  while (running) {
    ch = getch();
    switch (ch) {
    case 'q':
      running = false;
      Time::quit();
      break;
    case KEY_RESIZE:
      TUI::drawLayout();
      TUI::refreshTrack();
      break;
    }
  }
  endwin();
}

void TUI::drawLayout() {
  wclear(stdscr);
  refresh();

  switch (session_.getSubject().getColor()) {
  case 0:
    init_pair(1, COLOR_BLACK, -1);
    break;
  case 1:
    init_pair(1, COLOR_RED, -1);
    break;
  case 2:
    init_pair(1, COLOR_GREEN, -1);
    break;
  case 3:
    init_pair(1, COLOR_YELLOW, -1);
    break;
  case 4:
    init_pair(1, COLOR_BLUE, -1);
    break;
  case 5:
    init_pair(1, COLOR_MAGENTA, -1);
    break;
  case 6:
    init_pair(1, COLOR_CYAN, -1);
    break;
  case 7:
    init_pair(1, COLOR_WHITE, -1);
    break;
  default:
    init_pair(1, COLOR_WHITE, -1);
    break;
  };

  attron(A_BOLD);
  mvprintw(1, (getWidth() - 5) / 2, "Flipd\n");
  attroff(A_BOLD);
  string title = " " + session_.getSubject().getName();

  int screenWidth = getWidth();
  vector<string> lines;
  string current;
  string word;

  for (int i = 0; i <= (int)title.size(); i++) {
    if (i == (int)title.size() || title[i] == ' ') {
      string candidate = current.empty() ? word : current + " " + word;
      if ((int)candidate.size() > screenWidth) {
        lines.push_back(current);
        current = word;
      } else {
        current = candidate;
      }
      word = "";
    } else {
      word += title[i];
    }
  }
  if (!current.empty())
    lines.push_back(current);

  attron(COLOR_PAIR(1));
  for (int i = 0; i < (int)lines.size(); i++) {
    mvprintw(2 + i,
             (screenWidth - utf8::distance(lines[i].begin(), lines[i].end())) /
                 2,
             "%s", lines[i].c_str());
  }
  attroff(COLOR_PAIR(1));
  refresh();

  // mvprintw(getHeight() - 2, 0, "%s", track);

  if (container == nullptr) {
    bool hasTrack = track[0] != '\0' && track[0] != '\n';
    int top_height = 2 + lines.size();
    int timer_height = 5;
    int bottom_height = hasTrack ? 2 : 0;
    int free_height = getHeight() - top_height - bottom_height;
    // if (free_height < timer_height + 2) {
    //   free_height = timer_height + 2;
    // }
    container = newwin(5, getWidth(), top_height + (free_height - timer_height) / 2, 0);
    wattron(container, COLOR_PAIR(1));
  } else {
    wresize(container, 5, getWidth());
    mvwin(container, ((getHeight() - 8) / 2) + 3, 0);
  }
  wrefresh(container);
}

void TUI::refreshTimeDrawing() {
  for (int i = 0; i < 5; i++) {
    wmove(container, i, 0);
    wclrtoeol(container);
    mvwprintw(container, i, 0, "%s", Time::time_[i].c_str());
  }
  wrefresh(container);
}

void TUI::listener() {
  char tmp[512] = {0};
  while (true) {
    memcpy(tmp, track, 512);
    FILE *pipe = popen("playerctl", "r");
    if (pipe == NULL) {
      log("Error executing playerctl");
      continue;
    }
    fgets(track, sizeof(track), pipe);
    pclose(pipe);
    if (track[0] != '\0' && memcmp(track, tmp, 512) != 0) {
      refreshTrack();
    }
    sleep(1);
  }
}

void TUI::refreshTrack() {
  string track_str(track);
  string content = " 󰎇 " + track_str + " ";

  int y = getHeight() - 2;
  int x = (getWidth() - utf8::distance(content.begin(), content.end()) - 2 ) / 2;
  move(y, x);
  clrtoeol();

  attron(COLOR_PAIR(10));
  mvprintw(y, x++, "");
  attron(COLOR_PAIR(11));
  mvprintw(y, x, "%s", content.c_str());
  x += utf8::distance(content.begin(), content.end());

  attron(COLOR_PAIR(10));
  mvprintw(y, x++, "");
  attrset(A_NORMAL);

  refresh();
}

int TUI::getHeight() { return getmaxy(stdscr); }

int TUI::getWidth() { return getmaxx(stdscr); }
